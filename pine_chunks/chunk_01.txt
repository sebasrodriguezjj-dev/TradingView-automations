f_size(_s) =>
    _s == "Tiny" ? size.tiny : _s == "Small" ? size.small : _s == "Large" ? size.large : size.normal

f_lineStyle(_s) =>
    _s == "DASHED" ? line.style_dashed : _s == "DOTTED" ? line.style_dotted : line.style_solid

f_signalBreakRef(_preference, _bull) =>
    _preference == "Wick" ? (_bull ? high : low) : close

f_trimLineLabelArrays(line[] _lines, label[] _labels, int _max) =>
    if array.size(_lines) > _max
        line.delete(array.shift(_lines))
        label.delete(array.shift(_labels))

f_trimLineArray(line[] _lines, int _max) =>
    if array.size(_lines) > _max
        line.delete(array.shift(_lines))

f_pushStructureMark(line[] _lines, label[] _labels, int _x1, float _y, string _txt, color _col, int _max) =>
    _line = line.new(_x1, _y, bar_index, _y, color=_col, width=structureLineWidth, style=f_lineStyle(structureLineStyleInput))
    _mid = int(math.floor((_x1 + bar_index) * 0.5))
    _label = label.new(_mid, _y, _txt, style=label.style_none, textcolor=_col, size=f_size(tagSizeInput))
    array.push(_lines, _line)
    array.push(_labels, _label)
    f_trimLineLabelArrays(_lines, _labels, _max)

        _oldLabel = array.shift(_labels)

            label.delete(_oldLabel)

        array.shift(_types)
