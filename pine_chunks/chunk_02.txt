f_pushBox(box[] _boxes, label[] _labels, float[] _tops, float[] _bots, int _left, float _top, float _bot, string _txt, color _bg, color _border, int _extend, bool _showLabel, int _max, int _fillOpacity, int _borderWidth) =>
    _box = box.new(_left, _top, bar_index + _extend, _bot, bgcolor=color.new(_bg, _fillOpacity), border_color=color.new(_border, 10), border_width=_borderWidth)
    array.push(_boxes, _box)
    array.push(_tops, _top)
    array.push(_bots, _bot)
    label _label = na
    if _showLabel
        _label := label.new(bar_index + _extend, (_top + _bot) * 0.5, _txt, style=label.style_label_left, color=color.new(_border, 8), textcolor=color.white, size=f_size(tagSizeInput))
    array.push(_labels, _label)
    if array.size(_boxes) > _max
        _oldBox = array.shift(_boxes)
        if not na(_oldBox)
            box.delete(_oldBox)
        _oldLabel = array.shift(_labels)
        if not na(_oldLabel)
            label.delete(_oldLabel)
        array.shift(_tops)
        array.shift(_bots)

    array.remove(_tops, _idx)
    array.remove(_bots, _idx)

    _c = request.security(syminfo.tickerid, _tf, close, barmerge.gaps_off, barmerge.lookahead_off)

    _lo = request.security(syminfo.tickerid, _tf, ta.lowest(low, 20), barmerge.gaps_off, barmerge.lookahead_off)
    _mid = (_hi + _lo) * 0.5
    int _bias = 0
    if _c > _mid
        _bias := 1
    else if _c < _mid
        _bias := -1
    _bias
