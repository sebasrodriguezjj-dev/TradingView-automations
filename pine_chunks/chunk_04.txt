nalBearBOS = useSMC and not na(lastInternalLow) and close < lastInternalLow and close[1] >= lastInternalLow
internalBullCHoCH = internalState == -1 and internalBullBOS
internalBearCHoCH = internalState == 1 and internalBearBOS

if internalBullBOS
    if showInternalMarks and showExtraVisuals
        f_pushStructureMark(structureLines, structureLabels, lastInternalHighBar, lastInternalHigh, internalBullCHoCH ? "iCHOCH" : "iBOS", color.new(colBull, 20), maxStructureMarks)
    internalState := 1

if internalBearBOS
    if showInternalMarks and showExtraVisuals
        f_pushStructureMark(structureLines, structureLabels, lastInternalLowBar, lastInternalLow, internalBearCHoCH ? "iCHOCH" : "iBOS", color.new(colBear, 20), maxStructureMarks)
    internalState := -1

eqTol = atr * eqToleranceAtr
newEqHigh = useSMC and showEQ and not na(swingPH) and not na(lastSwingHigh) and not na(prevSwingHigh) and math.abs(lastSwingHigh - prevSwingHigh) <= eqTol
newEqLow = useSMC and showEQ and not na(swingPL) and not na(lastSwingLow) and not na(prevSwingLow) and math.abs(lastSwingLow - prevSwingLow) <= eqTol

if newEqHigh and showExtraVisuals
    _eqLine = line.new(lastSwingHighBar, lastSwingHigh, bar_index + 40, lastSwingHigh, extend=extend.right, color=color.new(colBear, 20), width=1, style=line.style_dashed)
    _eqLabel = label.new(bar_index + 40, lastSwingHigh, "EQH", style=label.style_label_left, color=color.new(colBear, 8), textcolor=color.white, size=f_size(tagSizeInput))

if newEqLow and showExtraVisuals
    _eqLine = line.new(lastSwingLowBar, lastSwingLow, bar_index + 40, lastSwingLow, extend=extend.right, color=color.new(colBull, 20), width=1, style=line.style_dashed)
    _eqLabel = label.new(bar_index + 40, lastSwingLow, "EQL", style=label.style_label_left, color=color.new(colBull, 8), textcolor=color.white, size=f_size(tagSizeInput))
    array.push(eqLines, _eqLine)
    array.push(eqLabels, _eqLabel)
    f_trimLineLabelArrays(eqLines, eqLabels, maxEQMarks)

recentHigh = useLiquidity ? ta.highest(high[1], sweepLookback) : na
recentLow = useLiquidity ? ta.lowest(low[1], sweepLookback) : na
rawBullSweep = useLiquidity and low < recentLow and close > recentLow and close > open and closePos > 0.55
rawBearSweep = useLiquidity and high > recentHigh and close < recentHigh and close < open and closePos < 0.45
bullPDSweep = useLiquidity and low < pdl and close > pdl
bearPDSweep = useLiquidity and high > pdh and close < pdh

levelBullSweep = false
levelBearSweep = false

if useLiquidity and showPivotSweeps and showExtraVisuals and array.size(liquidityLevelLines) > 0
    for i = array.size(liquidityLevelLines) - 1 to 0
        _level = array.get(liquidityLevels, i)
        _type = array.get(liquidityTypes, i)
        _start = array.get(liquidityStartBars, i)
        if _type == 1
            if high > _level and close < _level
                levelBearSweep := true
                if showSweepMarkers
                    f_pushStructureMark(liquidityEventLines, liquidityEventLabels, _start, _level, "BSL SWEEP", colBear, maxStructureMarks)
                f_deleteLiquidityLevelAt(liquidityLevelLines, liquidityLevelLabels, liquidityLevels, liquidityTypes, liquidityStartBars, i)
            else if close > _level
                f_deleteLiquidityLevelAt(liquidityLevelLines, liquidityLevelLabels, liquidityLevels, liquidityTypes, liquidityStartBars, i)
        else
            if low < _level and close > _level
                levelBullSweep := true
                if showSweepMarkers
                    f_pushStructureMark(liquidityEventLines, liquidityEventLabels, _start, _level, "SSL SWEEP", colBull, maxStructureMarks)
                f_deleteLiquidityLevelAt(liquidityLevelLines, liquidityLevelLabels, liquidityLevels, liquidityTypes, liquidityStartBars, i)
            else if close < _level
                f_deleteLiquidityLevelAt(liquidityLevelLines, liquidityLevelLabels, liquidityLevels, liquidityTypes, liquidityStartBars, i)

bullSweep = (showLookbackSweeps and rawBullSweep) or (showPivotSweeps and levelBullSweep)
bearSweep = (showLookbackSweeps and rawBearSweep) or (showPivotSweeps and levelBearSweep)
